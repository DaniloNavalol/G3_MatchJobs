Parte 3: Frontend (HTML e CSS)

Esta seção documenta a estrutura visual e o design da aplicação, cobrindo todos os arquivos HTML que definem o esqueleto das páginas e os arquivos CSS que aplicam os estilos.

1. Páginas HTML

Arquivos que definem a estrutura e o conteúdo de cada página do site.

index.html

/Codigo/site/index.html

<!DOCTYPE html> <-- Define o tipo de documento como HTML5 -->
<html lang="pt-br"> <-- Inicia o documento HTML, definindo o idioma como português do Brasil -->
<head> <-- Seção de metadados do documento -->
    <meta charset="UTF-8"> <-- Define o conjunto de caracteres como UTF-8 para suportar acentuação -->
    <meta name="viewport" content="width=device-width, initial-scale=1.0"> <-- Configura a viewport para um design responsivo, adaptando a largura ao dispositivo -->
    <link rel="stylesheet" href="css/style-index.css"> <-- Vincula a folha de estilos CSS externa para esta página -->
    <title>MatchJobs</title> <-- Define o título que aparece na aba do navegador -->
</head> <-- Fim da seção de metadados -->
<body> <-- Início do corpo do documento, conteúdo visível da página -->
    <header> <-- Define o cabeçalho da página -->
        <h1>MatchJobs</h1> <-- Título principal da página, geralmente o nome do site -->
        <nav> <-- Define uma seção de navegação -->
            <a href="pages/login.html"><button>Login</button></a> <-- Um link para a página de login, estilizado como um botão -->
            <a href="pages/cadastro.html"><button>Cadastro</button></a> <-- Um link para a página de cadastro, estilizado como um botão -->
        </nav> <-- Fim da seção de navegação -->
    </header> <-- Fim do cabeçalho -->
    <main> <-- Define o conteúdo principal da página -->
        <section> <-- Define uma seção dentro do conteúdo principal -->
            <div class="texto"> <-- Um container para o texto de chamada -->
                <h2>Encontre o emprego dos seus sonhos!</h2> <-- Subtítulo principal da seção -->
                <h3>A plataforma que conecta você às melhores oportunidades.</h3> <-- Texto de apoio -->
            </div> <-- Fim do container de texto -->
            <div class="imagem"> <-- Um container para a imagem de destaque -->
                <img src="img/img1.png" alt="Imagem Ilustrativa"> <-- Imagem principal, com texto alternativo para acessibilidade -->
            </div> <-- Fim do container de imagem -->
        </section> <-- Fim da seção -->
    </main> <-- Fim do conteúdo principal -->
</body> <-- Fim do corpo do documento -->
</html> <-- Fim do documento HTML -->

login.html

/Codigo/site/pages/login.html

<!DOCTYPE html>
<html lang="pt-br">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <link rel="stylesheet" href="../css/style-login.css"> <-- Vincula a folha de estilos específica para a página de login -->
    <title>Login</title>
</head>
<body>
    <div class="login-container"> <-- Container principal para o formulário de login -->
        <h2>Login</h2> <-- Título do formulário -->
        <form id="login-form"> <-- Início do formulário de login. O ID é usado pelo JavaScript para manipulação -->
            <div class="input-group"> <-- Agrupa um rótulo e um campo de entrada -->
                <label for="email">Email:</label> <-- Rótulo para o campo de email -->
                <input type="email" id="email" name="email" required> <-- Campo de entrada para o email, do tipo 'email' para validação básica e 'required' para torná-lo obrigatório -->
            </div>
            <div class="input-group">
                <label for="senha">Senha:</label> <-- Rótulo para o campo de senha -->
                <input type="password" id="senha" name="senha" required> <-- Campo de entrada para a senha, do tipo 'password' para mascarar os caracteres -->
            </div>
            <div class="radio-group"> <-- Agrupa os botões de rádio para seleção do tipo de usuário -->
                <input type="radio" id="candidato" name="tipo_usuario" value="candidato" checked> <-- Botão de rádio para 'Candidato', pré-selecionado (checked) -->
                <label for="candidato">Candidato</label>
                <input type="radio" id="empresa" name="tipo_usuario" value="empresa"> <-- Botão de rádio para 'Empresa' -->
                <label for="empresa">Empresa</label>
            </div>
            <button type="submit">Entrar</button> <-- Botão para submeter o formulário -->
            <p id="error-message" style="color: red; display: none;"></p> <-- Parágrafo para exibir mensagens de erro, inicialmente oculto -->
        </form>
        <p>Não tem uma conta? <a href="cadastro.html">Cadastre-se</a></p> <-- Link para a página de cadastro para novos usuários -->
    </div>
    <script src="../js/script-login.js"></script> <-- Vincula o arquivo JavaScript que contém a lógica do formulário de login -->
</body>
</html>

2. Estilos CSS

Folhas de estilo que definem a aparência de cada página.

style-login.css

/Codigo/site/css/style-login.css

body { /* Estiliza o corpo da página */
    font-family: Arial, sans-serif; /* Define a família de fontes */
    background-color: #f4f4f4; /* Define a cor de fundo */
    display: flex; /* Habilita o layout Flexbox */
    justify-content: center; /* Centraliza o conteúdo horizontalmente */
    align-items: center; /* Centraliza o conteúdo verticalmente */
    height: 100vh; /* Define a altura total da tela */
    margin: 0; /* Remove a margem padrão */
}

.login-container { /* Estiliza o container do formulário */
    background-color: white; /* Fundo branco */
    padding: 2rem; /* Espaçamento interno */
    border-radius: 5px; /* Cantos arredondados */
    box-shadow: 0 0 10px rgba(0, 0, 0, 0.1); /* Sombra sutil */
    width: 300px; /* Largura fixa */
}

h2 { /* Estiliza os títulos de nível 2 */
    text-align: center; /* Texto centralizado */
    margin-bottom: 1.5rem; /* Margem inferior */
}

.input-group { /* Estiliza o grupo de rótulo e campo */
    margin-bottom: 1rem; /* Margem inferior para espaçamento */
}

.input-group label { /* Estiliza os rótulos */
    display: block; /* Faz o rótulo ocupar uma linha inteira */
    margin-bottom: 5px; /* Margem inferior pequena */
}

.input-group input { /* Estiliza os campos de entrada */
    width: 100%; /* Largura total do container pai */
    padding: 8px; /* Espaçamento interno */
    box-sizing: border-box; /* Faz com que o padding não afete a largura total */
}

.radio-group { /* Estiliza o grupo de botões de rádio */
    margin-bottom: 1rem; /* Margem inferior */
}

button { /* Estiliza o botão de submit */
    width: 100%; /* Largura total */
    padding: 10px; /* Espaçamento interno */
    background-color: #5cb85c; /* Cor de fundo verde */
    color: white; /* Cor do texto branca */
    border: none; /* Remove a borda padrão */
    border-radius: 5px; /* Cantos arredondados */
    cursor: pointer; /* Muda o cursor para uma mão ao passar por cima */
}

button:hover { /* Estiliza o botão quando o mouse está sobre ele */
    background-color: #4cae4c; /* Cor de fundo verde mais escura */
}

p { /* Estiliza os parágrafos */
    text-align: center; /* Texto centralizado */
    margin-top: 1rem; /* Margem superior */
}
