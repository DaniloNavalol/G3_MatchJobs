64 package com.matchjobs.service; <span class="comment">// Define o pacote para a classe de serviço.</span>
65 
66 import java.util.Optional; <span class="comment">// Importa a classe Optional para tratar valores potencialmente nulos.</span>
67 import org.springframework.beans.factory.annotation.Autowired; <span class="comment">// Importa a anotação para injeção de dependência.</span>
68 import org.springframework.stereotype.Service; <span class="comment">// Importa a anotação que marca a classe como um serviço do Spring.</span>
69 import com.matchjobs.model.Empresa; <span class="comment">// Importa a entidade Empresa.</span>
70 import com.matchjobs.model.Login; <span class="comment">// Importa o DTO Login.</span>
71 import com.matchjobs.model.Usuario; <span class="comment">// Importa a entidade Usuario.</span>
72 import com.matchjobs.repository.EmpresaRepository; <span class="comment">// Importa o repositório de Empresa.</span>
73 import com.matchjobs.repository.UsuarioRepository; <span class="comment">// Importa o repositório de Usuario.</span>
74 
75 @Service <span class="comment">// Anota a classe como um componente de serviço, tornando-a elegível para injeção de dependência.</span>
76 public class LoginService { <span class="comment">// Inicia a declaração da classe pública 'LoginService'.</span>
77 
78  @Autowired <span class="comment">// Injeta uma instância de UsuarioRepository automaticamente.</span>
79  private UsuarioRepository usuarioRepository; <span class="comment">// Declara o campo para o repositório de usuário.</span>
80 
81  @Autowired <span class="comment">// Injeta uma instância de EmpresaRepository automaticamente.</span>
82  private EmpresaRepository empresaRepository; <span class="comment">// Declara o campo para o repositório de empresa.</span>
83 
84  public boolean validarLogin(Login login) { <span class="comment">// Define o método público para validar as credenciais de login.</span>
85  if ("candidato".equals(login.getTipoUsuario())) { <span class="comment">// Verifica se o tipo de usuário no DTO é "candidato".</span>
86  Optional&lt;Usuario&gt; usuarioOpt = usuarioRepository.findByEmail(login.getEmail()); <span class="comment">// Busca o usuário no banco de dados pelo email fornecido.</span>
87  // ATENÇÃO: A comparação de senhas em texto plano é uma falha de segurança grave.
88  // Em um sistema real, use um encoder (ex: BCrypt) para comparar o hash da senha.
89  return usuarioOpt.isPresent() && usuarioOpt.get().getSenha().equals(login.getSenha()); <span class="comment">// Retorna true se o usuário existe E a senha corresponde.</span>
90  } else if ("empresa".equals(login.getTipoUsuario())) { <span class="comment">// Se não for candidato, verifica se o tipo é "empresa".</span>
91  Optional&lt;Empresa&gt; empresaOpt = empresaRepository.findByEmail(login.getEmail()); <span class="comment">// Busca a empresa no banco de dados pelo email fornecido.</span>
92  return empresaOpt.isPresent() && empresaOpt.get().getSenha().equals(login.getSenha()); <span class="comment">// Retorna true se a empresa existe E a senha corresponde.</span>
93  } <span class="comment">// Fim do bloco if.</span>
94  return false; <span class="comment">// Se o tipo de usuário não for nem "candidato" nem "empresa", retorna false.</span>
95  } <span class="comment">// Fim do método validarLogin.</span>
96 } <span class="comment">// Fim da classe LoginService.</span>

101  <h4>EmpresaService.java, UsuarioService.java, VagasService.java</h4>
102  <p>No código fornecido, as classes `EmpresaService`, `UsuarioService` e `VagasService` não foram criadas. A lógica de negócio está sendo chamada diretamente dos controllers para os repositórios. Em uma aplicação maior, seria ideal criar essas classes de serviço para encapsular as regras de negócio, como validações, e desacoplar os controllers do acesso direto aos dados.</p>

135  } else { <span class="comment">// Caso as credenciais sejam inválidas.</span>

137  } <span class="comment">// Fim do bloco if/else.</span>
138  } <span class="comment">// Fim do método login.</span>
139 } <span class="comment">// Fim da classe 'LoginController'.</span>
