99 document.addEventListener('DOMContentLoaded', function() { <span class="comment">// Adiciona um ouvinte ao documento que dispara a função quando todo o conteúdo HTML da página foi carregado e analisado.</span>
100  const vagasContainer = document.getElementById('vagas-container'); <span class="comment">// Obtém a referência ao elemento div com o ID 'vagas-container', onde os cards das vagas serão inseridos.</span>
101 
102  fetch('http://localhost:8080/vagas') <span class="comment">// Inicia uma requisição de rede do tipo GET (padrão) para a URL da API que lista as vagas.</span>
103  .then(response => { <span class="comment">// O '.then()' é executado quando o servidor responde.</span>
104  if (!response.ok) { <span class="comment">// Verifica se a resposta NÃO foi bem-sucedida (ex: status 404 ou 500).</span>
105  throw new Error('Erro na rede: ' + response.statusText); <span class="comment">// Se a resposta não foi OK, lança um novo erro, que será capturado pelo bloco '.catch()'.</span>
106  } <span class="comment">// Fim do if.</span>
107  return response.json(); <span class="comment">// Se a resposta foi OK, converte o corpo da resposta de JSON para um objeto/array JavaScript.</span>
108  }) <span class="comment">// Fim do primeiro '.then()'.</span>
109  .then(data => { <span class="comment">// Este '.then()' é executado com os dados já convertidos (o array de vagas).</span>
110  vagasContainer.innerHTML = ''; <span class="comment">// Limpa qualquer conteúdo existente dentro do container de vagas (ex: uma mensagem "Carregando...").</span>
111  data.forEach(vaga => { <span class="comment">// Itera sobre cada elemento ('vaga') dentro do array 'data'.</span>
112  const vagaElement = document.createElement('div'); <span class="comment">// Cria um novo elemento HTML 'div' na memória para cada vaga.</span>
113  vagaElement.classList.add('vaga'); <span class="comment">// Adiciona a classe CSS 'vaga' a este novo elemento 'div' para estilização.</span>
114 
115  vagaElement.innerHTML = ` <span class="comment">// Define o conteúdo HTML interno do novo 'div' usando uma template string.</span>
116  &lt;h3&gt;${vaga.nome}&lt;/h3&gt;
117  &lt;p&gt;&lt;strong&gt;Senioridade:&lt;/strong&gt; ${vaga.senioridade}&lt;/p&gt;
118  &lt;p&gt;&lt;strong&gt;Salário:&lt;/strong&gt; ${vaga.salario}&lt;/p&gt;
119  &lt;p&gt;${vaga.descricao}&lt;/p&gt;
120  `; <span class="comment">// Fim da definição do HTML interno. As variáveis como ${vaga.nome} são injetadas na string.</span>
121 
122  vagasContainer.appendChild(vagaElement); <span class="comment">// Adiciona o novo elemento 'div' (o card da vaga) como um filho do 'vagasContainer', tornando-o visível na página.</span>
123  }); <span class="comment">// Fim do forEach.</span>
124  }) <span class="comment">// Fim do segundo '.then()'.</span>
125  .catch(error => { <span class="comment">// Captura erros de rede ou o erro lançado manualmente no primeiro '.then()'.</span>
126  console.error('Erro ao buscar vagas:', error); <span class="comment">// Exibe o erro no console do navegador para depuração.</span>
127  vagasContainer.innerHTML = '&lt;p&gt;Não foi possível carregar as vagas. Tente novamente mais tarde.&lt;/p&gt;'; <span class="comment">// Exibe uma mensagem de erro amigável para o usuário dentro do container.</span>
128  }); <span class="comment">// Fim do '.catch()'.</span>
129 }); <span class="comment">// Fim da função do ouvinte de evento 'DOMContentLoaded'.</span>
