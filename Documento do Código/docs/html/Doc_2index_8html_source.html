211  <h2 class="title"> Contact Form </h2>
212  <p>This is a PHP Script for sending messages to your email, you should replace <strong>demo@site.com</strong> to your email to start receive messages.</p>
213  <pre class="brush:php">
214  $to = 'demo@site.com'; // replace this mail with yours
215  $firstname = $_POST["fname"];
216  $email= $_POST["email"];
217  $headers = 'MIME-Version: 1.0' . "\r\n";
218  $headers .= "From: " . $email . "\r\n"; // Sender's E-mail
219  $headers .= 'Content-type: text/html; charset=iso-8859-1' . "\r\n";
220 
221  $message ='<table style="width:100%">
222  <tr>
223  <td>'.$firstname.' '.$laststname.'</td>
224  </tr>
225  <tr><td>Email: '.$email.'</td></tr>
226  </table>';
227 
228  if (@mail($to, $email, $message, $headers))
229  {
230  echo 'The message has been sent.';
231  }else{
232  echo 'failed';
233  }
