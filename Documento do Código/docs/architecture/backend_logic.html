Parte 2: Services e Controllers

Esta seção documenta a lógica de negócios (`Services`) que interage com os repositórios e a camada de API (`Controllers`) que expõe essa lógica ao mundo exterior através de endpoints HTTP.

1. Services

Classes que contêm as regras de negócio da aplicação.

LoginService.java

/Codigo/MatchJobs/src/main/java/com/matchjobs/service/LoginService.java

package com.matchjobs.service; // Define o pacote para a classe de serviço.

import java.util.Optional; // Importa a classe Optional para tratar valores potencialmente nulos.
import org.springframework.beans.factory.annotation.Autowired; // Importa a anotação para injeção de dependência.
import org.springframework.stereotype.Service; // Importa a anotação que marca a classe como um serviço do Spring.
import com.matchjobs.model.Empresa; // Importa a entidade Empresa.
import com.matchjobs.model.Login; // Importa o DTO Login.
import com.matchjobs.model.Usuario; // Importa a entidade Usuario.
import com.matchjobs.repository.EmpresaRepository; // Importa o repositório de Empresa.
import com.matchjobs.repository.UsuarioRepository; // Importa o repositório de Usuario.

@Service // Anota a classe como um componente de serviço, tornando-a elegível para injeção de dependência.
public class LoginService { // Inicia a declaração da classe pública 'LoginService'.

    @Autowired // Injeta uma instância de UsuarioRepository automaticamente.
    private UsuarioRepository usuarioRepository; // Declara o campo para o repositório de usuário.

    @Autowired // Injeta uma instância de EmpresaRepository automaticamente.
    private EmpresaRepository empresaRepository; // Declara o campo para o repositório de empresa.

    public boolean validarLogin(Login login) { // Define o método público para validar as credenciais de login.
        if ("candidato".equals(login.getTipoUsuario())) { // Verifica se o tipo de usuário no DTO é "candidato".
            Optional<Usuario> usuarioOpt = usuarioRepository.findByEmail(login.getEmail()); // Busca o usuário no banco de dados pelo email fornecido.
            // ATENÇÃO: A comparação de senhas em texto plano é uma falha de segurança grave.
            // Em um sistema real, use um encoder (ex: BCrypt) para comparar o hash da senha.
            return usuarioOpt.isPresent() && usuarioOpt.get().getSenha().equals(login.getSenha()); // Retorna true se o usuário existe E a senha corresponde.
        } else if ("empresa".equals(login.getTipoUsuario())) { // Se não for candidato, verifica se o tipo é "empresa".
            Optional<Empresa> empresaOpt = empresaRepository.findByEmail(login.getEmail()); // Busca a empresa no banco de dados pelo email fornecido.
            return empresaOpt.isPresent() && empresaOpt.get().getSenha().equals(login.getSenha()); // Retorna true se a empresa existe E a senha corresponde.
        } // Fim do bloco if.
        return false; // Se o tipo de usuário não for nem "candidato" nem "empresa", retorna false.
    } // Fim do método validarLogin.
} // Fim da classe LoginService.

EmpresaService.java, UsuarioService.java, VagasService.java

No código fornecido, as classes `EmpresaService`, `UsuarioService` e `VagasService` não foram criadas. A lógica de negócio está sendo chamada diretamente dos controllers para os repositórios. Em uma aplicação maior, seria ideal criar essas classes de serviço para encapsular as regras de negócio, como validações, e desacoplar os controllers do acesso direto aos dados.

2. Controllers

Classes que expõem os endpoints da API REST.

LoginController.java

/Codigo/MatchJobs/src/main/java/com/matchjobs/controller/LoginController.java

package com.matchjobs.controller; // Define o pacote para a classe de controller.

import org.springframework.beans.factory.annotation.Autowired; // Importa a anotação para injeção de dependência.
import org.springframework.http.ResponseEntity; // Importa a classe para construir respostas HTTP.
import org.springframework.web.bind.annotation.*; // Importa anotações do Spring Web.
import com.matchjobs.model.Login; // Importa o DTO Login.
import com.matchjobs.service.LoginService; // Importa a classe de serviço de Login.

@RestController // Marca a classe como um controlador REST.
@RequestMapping("/login") // Mapeia as requisições para o caminho base "/login".
@CrossOrigin(origins = "*") // Habilita o CORS para permitir requisições de qualquer origem.
public class LoginController { // Inicia a declaração da classe pública 'LoginController'.

    @Autowired // Injeta uma instância de 'LoginService' automaticamente.
    private LoginService loginService; // Declara o campo para o serviço de login.

    @PostMapping // Mapeia este método para requisições HTTP do tipo POST para "/login".
    public ResponseEntity<Void> login(@RequestBody Login login) { // Define o método de login que recebe um objeto 'Login' do corpo da requisição.
        if (loginService.validarLogin(login)) { // Chama o serviço para validar as credenciais.
            return ResponseEntity.ok().build(); // Se válido, retorna uma resposta com status 200 (OK) e corpo vazio.
        } else { // Caso as credenciais sejam inválidas.
            return ResponseEntity.status(401).build(); // Retorna uma resposta com status 401 (Não Autorizado) e corpo vazio.
        } // Fim do bloco if/else.
    } // Fim do método login.
} // Fim da classe 'LoginController'.

VagasController.java

/Codigo/MatchJobs/src/main/java/com/matchjobs/controller/VagasController.java

package com.matchjobs.controller; // Define o pacote para a classe de controller.

import java.util.List; // Importa a interface List para trabalhar com coleções.
import org.springframework.beans.factory.annotation.Autowired; // Importa a anotação para injeção de dependência.
import org.springframework.web.bind.annotation.CrossOrigin; // Importa a anotação para configurar o CORS.
import org.springframework.web.bind.annotation.GetMapping; // Importa a anotação para mapear requisições HTTP GET.
import org.springframework.web.bind.annotation.PostMapping; // Importa a anotação para mapear requisições HTTP POST.
import org.springframework.web.bind.annotation.RequestBody; // Importa a anotação para vincular o corpo da requisição a um objeto.
import org.springframework.web.bind.annotation.RequestMapping; // Importa a anotação para mapear o caminho base do controller.
import org.springframework.web.bind.annotation.RestController; // Importa a anotação que marca a classe como um controlador REST.
import com.matchjobs.model.Vagas; // Importa a entidade Vagas.
import com.matchjobs.repository.VagasRepository; // Importa o repositório de Vagas.

@RestController // Marca a classe como um controlador REST.
@RequestMapping("/vagas") // Mapeia as requisições para o caminho base "/vagas".
@CrossOrigin(origins = "*") // Habilita o CORS.
public class VagasController { // Inicia a declaração da classe pública 'VagasController'.
    
    @Autowired // Injeta uma instância de 'VagasRepository' automaticamente.
    private VagasRepository vagasRepository; // Declara o campo para o repositório de vagas.

    @GetMapping // Mapeia este método para requisições HTTP GET para "/vagas".
    public List<Vagas> listar() { // Define o método que retorna uma lista de vagas.
        return vagasRepository.findAll(); // Chama o método 'findAll' do repositório para buscar todas as vagas no banco.
    } // Fim do método listar.
    
    @PostMapping // Mapeia este método para requisições HTTP POST para "/vagas".
    public Vagas cadastrar(@RequestBody Vagas vaga) { // Define o método que recebe um objeto 'Vagas' do corpo da requisição.
        return vagasRepository.save(vaga); // Chama o método 'save' do repositório para salvar a nova vaga no banco.
    } // Fim do método cadastrar.
} // Fim da classe 'VagasController'.
