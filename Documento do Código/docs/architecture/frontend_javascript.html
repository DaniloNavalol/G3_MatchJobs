Parte 4: Frontend (JavaScript)

Esta seção final documenta a camada de script do lado do cliente, detalhando os arquivos JavaScript responsáveis pela lógica dos formulários, requisições à API e manipulação dinâmica do conteúdo das páginas.

1. Scripts JavaScript

script-login.js

/Codigo/site/js/script-login.js

document.getElementById('login-form').addEventListener('submit', function(event) { // Seleciona o elemento do formulário pelo seu ID 'login-form' e adiciona um "ouvinte" para o evento 'submit'.
    event.preventDefault(); // Previne o comportamento padrão do formulário, que seria recarregar a página ao submeter.

    const email = document.getElementById('email').value; // Declara uma constante 'email' e armazena nela o valor atual do campo com ID 'email'.
    const senha = document.getElementById('senha').value; // Declara uma constante 'senha' e armazena nela o valor atual do campo com ID 'senha'.
    const tipoUsuario = document.querySelector('input[name="tipo_usuario"]:checked').value; // Seleciona o botão de rádio com o nome 'tipo_usuario' que está marcado (:checked) e obtém seu valor.
    const mensagemErro = document.getElementById('error-message'); // Obtém a referência ao elemento de parágrafo com ID 'error-message' para exibir erros.

    const data = { // Cria um objeto literal JavaScript para organizar os dados que serão enviados ao backend.
        email: email, // Adiciona a propriedade 'email' com o valor da constante 'email'.
        senha: senha, // Adiciona a propriedade 'senha' com o valor da constante 'senha'.
        tipoUsuario: tipoUsuario // Adiciona a propriedade 'tipoUsuario' com o valor obtido do botão de rádio.
    }; // Fim da criação do objeto.

    fetch('http://localhost:8080/login', { // Inicia uma requisição de rede para a URL da API de login usando a Fetch API.
        method: 'POST', // Define o método HTTP da requisição como 'POST', para enviar dados.
        headers: { // Define os cabeçalhos da requisição.
            'Content-Type': 'application/json' // Informa ao servidor que o corpo da requisição está no formato JSON.
        }, // Fim da definição dos cabeçalhos.
        body: JSON.stringify(data) // Converte o objeto JavaScript 'data' em uma string no formato JSON para ser enviada no corpo da requisição.
    }) // Fim da configuração da requisição fetch.
    .then(response => { // O método '.then()' é executado quando o servidor envia uma resposta. 'response' contém os dados da resposta.
        if (response.ok) { // A propriedade 'ok' da resposta é 'true' se o status HTTP for bem-sucedido (ex: 200).
            localStorage.setItem('tipoUsuario', tipoUsuario); // Se o login for bem-sucedido, armazena o tipo de usuário no armazenamento local (localStorage) do navegador.
            window.location.href = 'vagas.html'; // Redireciona o navegador do usuário para a página 'vagas.html'.
        } else { // Se a propriedade 'ok' for 'false' (ex: erro 401).
            mensagemErro.textContent = 'Email ou senha inválidos.'; // Define o texto do elemento de erro.
            mensagemErro.style.display = 'block'; // Muda o estilo do elemento de erro para 'block' para torná-lo visível na página.
        } // Fim do bloco if/else.
    }) // Fim do primeiro '.then()'.
    .catch(error => { // O método '.catch()' é executado se ocorrer um erro na requisição (ex: falha de rede).
        console.error('Erro no login:', error); // Exibe o objeto de erro no console do navegador para fins de depuração.
        mensagemErro.textContent = 'Erro ao tentar fazer login. Tente novamente mais tarde.'; // Exibe uma mensagem de erro mais genérica para o usuário.
        mensagemErro.style.display = 'block'; // Torna a mensagem de erro visível.
    }); // Fim do '.catch()'.
}); // Fim da função do ouvinte de evento 'submit'.

script-vagas.js

/Codigo/site/js/script-vagas.js

document.addEventListener('DOMContentLoaded', function() { // Adiciona um ouvinte ao documento que dispara a função quando todo o conteúdo HTML da página foi carregado e analisado.
    const vagasContainer = document.getElementById('vagas-container'); // Obtém a referência ao elemento div com o ID 'vagas-container', onde os cards das vagas serão inseridos.

    fetch('http://localhost:8080/vagas') // Inicia uma requisição de rede do tipo GET (padrão) para a URL da API que lista as vagas.
        .then(response => { // O '.then()' é executado quando o servidor responde.
            if (!response.ok) { // Verifica se a resposta NÃO foi bem-sucedida (ex: status 404 ou 500).
                throw new Error('Erro na rede: ' + response.statusText); // Se a resposta não foi OK, lança um novo erro, que será capturado pelo bloco '.catch()'.
            } // Fim do if.
            return response.json(); // Se a resposta foi OK, converte o corpo da resposta de JSON para um objeto/array JavaScript.
        }) // Fim do primeiro '.then()'.
        .then(data => { // Este '.then()' é executado com os dados já convertidos (o array de vagas).
            vagasContainer.innerHTML = ''; // Limpa qualquer conteúdo existente dentro do container de vagas (ex: uma mensagem "Carregando...").
            data.forEach(vaga => { // Itera sobre cada elemento ('vaga') dentro do array 'data'.
                const vagaElement = document.createElement('div'); // Cria um novo elemento HTML 'div' na memória para cada vaga.
                vagaElement.classList.add('vaga'); // Adiciona a classe CSS 'vaga' a este novo elemento 'div' para estilização.

                vagaElement.innerHTML = ` // Define o conteúdo HTML interno do novo 'div' usando uma template string.
                    <h3>${vaga.nome}</h3>
                    <p><strong>Senioridade:</strong> ${vaga.senioridade}</p>
                    <p><strong>Salário:</strong> ${vaga.salario}</p>
                    <p>${vaga.descricao}</p>
                `; // Fim da definição do HTML interno. As variáveis como ${vaga.nome} são injetadas na string.
                
                vagasContainer.appendChild(vagaElement); // Adiciona o novo elemento 'div' (o card da vaga) como um filho do 'vagasContainer', tornando-o visível na página.
            }); // Fim do forEach.
        }) // Fim do segundo '.then()'.
        .catch(error => { // Captura erros de rede ou o erro lançado manualmente no primeiro '.then()'.
            console.error('Erro ao buscar vagas:', error); // Exibe o erro no console do navegador para depuração.
            vagasContainer.innerHTML = '<p>Não foi possível carregar as vagas. Tente novamente mais tarde.</p>'; // Exibe uma mensagem de erro amigável para o usuário dentro do container.
        }); // Fim do '.catch()'.
}); // Fim da função do ouvinte de evento 'DOMContentLoaded'.
