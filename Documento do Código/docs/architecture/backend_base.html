package com.matchjobs.model; // Define que esta classe pertence ao pacote 'com.matchjobs.model'.

import javax.persistence.Entity; // Importa a anotação 'Entity' para mapeamento objeto-relacional.
import javax.persistence.GeneratedValue; // Importa a anotação 'GeneratedValue' para geração automática de chaves primárias.
import javax.persistence.GenerationType; // Importa os tipos de estratégias de geração de chaves.
import javax.persistence.Id; // Importa a anotação 'Id' para definir a chave primária.

@Entity(name = "empresa") // Anota a classe como uma entidade JPA e a mapeia para a tabela "empresa".
public class Empresa { // Inicia a declaração da classe pública 'Empresa'.

    @Id // Marca o campo 'id_empresa' como a chave primária da tabela.
    @GeneratedValue(strategy = GenerationType.IDENTITY) // Configura a geração do valor da chave primária para ser autoincremental pelo banco.
    private Long id_empresa; // Declara o campo para o ID da empresa.

    private String nome; // Declara o campo para o nome da empresa.
    private String cnpj; // Declara o campo para o CNPJ da empresa.
    private String email; // Declara o campo para o email da empresa.
    private String telefone; // Declara o campo para o telefone da empresa.
    private String senha; // Declara o campo para a senha da empresa.

    public Empresa() { // Declara o construtor padrão (vazio), exigido pelo JPA.
    } // Fim do construtor.

    public Long getId_empresa() { // Método getter para o ID da empresa.
        return id_empresa; // Retorna o valor do ID da empresa.
    } // Fim do método getId_empresa.

    public void setId_empresa(Long id_empresa) { // Método setter para o ID da empresa.
        this.id_empresa = id_empresa; // Atribui o valor passado ao campo 'id_empresa' da instância.
    } // Fim do método setId_empresa.

    public String getNome() { // Método getter para o nome.
        return nome; // Retorna o valor do nome.
    } // Fim do método getNome.

    public void setNome(String nome) { // Método setter para o nome.
        this.nome = nome; // Atribui o valor ao campo 'nome'.
    } // Fim do método setNome.
    
    public String getCnpj() { // Método getter para o CNPJ.
        return cnpj; // Retorna o valor do CNPJ.
    } // Fim do método getCnpj.

    public void setCnpj(String cnpj) { // Método setter para o CNPJ.
        this.cnpj = cnpj; // Atribui o valor ao campo 'cnpj'.
    } // Fim do método setCnpj.

    public String getEmail() { // Método getter para o email.
        return email; // Retorna o valor do email.
    } // Fim do método getEmail.

    public void setEmail(String email) { // Método setter para o email.
        this.email = email; // Atribui o valor ao campo 'email'.
    } // Fim do método setEmail.

    public String getTelefone() { // Método getter para o telefone.
        return telefone; // Retorna o valor do telefone.
    } // Fim do método getTelefone.

    public void setTelefone(String telefone) { // Método setter para o telefone.
        this.telefone = telefone; // Atribui o valor ao campo 'telefone'.
    } // Fim do método setTelefone.

    public String getSenha() { // Método getter para a senha.
        return senha; // Retorna o valor da senha.
    } // Fim do método getSenha.

    public void setSenha(String senha) { // Método setter para a senha.
        this.senha = senha; // Atribui o valor ao campo 'senha'.
    } // Fim do método setSenha.

} // Fim da classe 'Empresa'.

package com.matchjobs.repository; // Define que esta interface pertence ao pacote 'com.matchjobs.repository'.

import java.util.Optional; // Importa a classe Optional para lidar com valores que podem ser nulos.
import org.springframework.data.jpa.repository.JpaRepository; // Importa a interface JpaRepository do Spring Data.
import org.springframework.stereotype.Repository; // Importa a anotação que marca a interface como um componente de repositório do Spring.
import com.matchjobs.model.Empresa; // Importa a entidade Empresa.

@Repository // Anota a interface como um repositório Spring, para ser gerenciada pelo contêiner.
public interface EmpresaRepository extends JpaRepository<Empresa, Long>{ // Declara a interface, que herda métodos CRUD de JpaRepository para a entidade Empresa com chave Long.

    Optional<Empresa> findByEmail(String email); // Define um método customizado para buscar uma Empresa pelo seu email. O Spring Data implementa este método automaticamente.
} // Fim da interface 'EmpresaRepository'.
